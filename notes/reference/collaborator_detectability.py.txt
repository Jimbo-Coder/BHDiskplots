def smart_genfromtxt(file_path):
  data = []
  with open(file_path,"r") as f:
    for line in f:
      li=line.strip()
      if not li.startswith("#"):
        data_line = line.split()[:27]
        data.append(data_line)
  data = np.array(data,dtype=float)
  return data

def LISA_sens_curve(f):
  # assume f is in Hz
  P_OMS = (1.5*10**(-11))**2 # in m^2 Hz^{-1}
  P_ACC = (3.0*10**(-15))**2 * (1 + (0.4*0.001/f)**2) # in m^2 s^{-4} Hz^{-1}
  A = 9.0E-45 
  # noise parameters for S_c, the confusion noise, are taken for a 4 yr observation time
  # from Table 1. 
  alpha = 0.138
  beta = -221
  kappa = 521
  gamma = 1680
  f_k = 0.00113
  S_c = A * f**(-7.0/3)*np.exp(-f**alpha + beta*f*np.sin(kappa*f))*(1 + np.tanh(gamma*(f_k - f))) # in Hz^{-1}
  L = 2.5 * 10**9 # m
  f_star = 0.001*19.09
  S_n = 10.0/(3*L**2)*(P_OMS * 4*P_ACC/(2*np.pi*f)**4)*(1 + 0.6*(f/f_star)**2) #+ S_c # in Hz^{-1}
  S_0 = np.sqrt(P_OMS * 10.0/(3*L**2))
  f_1 = (4*(3.0*10**(-15))**2/P_OMS)**(0.25)/(2*np.pi)
  f_2 = (f_1**4 * 0.0004**2)**(1.0/6.0)
  f_3 = f_star/np.sqrt(0.6)
  #print(S_0)
  #print("f_1 = ",f_1)
  #print("f_2 = ",f_2)
  #print("f_3 = ",f_3)
  # sqrt(S_n) = S_0 * np.sqrt( (1 + (f_1/f)**4 + (f_2/f)**6 )*( 1 + (f/f_3)**2 ) ) 
  # h_n = np.sqrt(S_n)
  #
  ASD_LISA = np.sqrt(2) * S_0 * np.sqrt( (1 + (f_1/f)**4 + (f_2/f)**6 )*( 1 + (f/f_3)**2 ) + S_c/S_0**2)
  return ASD_LISA

def get_fft(rhp,rhc,npadding,dt,alpha):
  fs = 1.0/dt
  f_Nyquist = fs/2.0
  padding = np.zeros(npadding)
  window = signal.windows.tukey(len(rhp),alpha) # Tukay signal window for the FFT
  hp_windowed_padded = np.concatenate((padding,rhp*window,padding))
  hc_windowed_padded = np.concatenate((padding,rhc*window,padding))
  Nt = len(hp_windowed_padded)
  f = np.linspace(0.0, (Nt-1)/Nt*fs, Nt)
  FFT_hp = np.abs(scipy.fftpack.fft(hp_windowed_padded))*dt
  FFT_hc = np.abs(scipy.fftpack.fft(hc_windowed_padded))*dt  
  FFT_h_total = (0.5*(FFT_hp**2+FFT_hc**2))**(0.5)
  N_max = int(Nt/2.0)
  f = f
  FFT_h_total = FFT_h_total
  return (f[2:N_max],FFT_h_total[2:N_max])

def get_Psi_full(data,theta,phi):
  rPsi_Re_full = np.zeros(len(data[:,0]))
  rPsi_Im_full = np.zeros(len(data[:,0]))
  r_ext = data[0,-4]
  n_mode = 0
  for ll in [2,3,4,5,6,7,8]:
    for im in range(0,2*ll+1):
      mm = ll - im
      rPsi_Re = data[:,1+2*n_mode]
      rPsi_Im = data[:,2+2*n_mode]
      rPsi = rPsi_Re + 1j*rPsi_Im
      S = sph.harmonic(-2, ll, mm, 0.0)
      #rh_plus_full += np.real(special.sph_harm(mm,ll,theta,phi))*rh_plus
      #rh_cross_full += np.real(special.sph_harm(mm,ll,theta,phi))*rh_plus
      rPsi_Re_full += np.real(S(theta,phi)*rPsi)
      rPsi_Im_full += np.imag(S(theta,phi)*rPsi)
      #rh_mag = np.sqrt(rh_plus**2 + rh_cross**2)
      n_mode += 1
  rPsi_Re_full *= r_ext
  rPsi_Im_full *= r_ext
  return (rPsi_Re_full,rPsi_Im_full)

def get_fft_Psi_full(data,theta,phi,alpha,npadding):
  rPsi_Re_full,rPsi_Im_full = get_Psi_full(data,theta,phi)
  dt = (data[1,0] - data[0,0])  # dt=(tb-ta)/(Nt-1)
  f, fft_Psi = get_fft(rPsi_Re_full,rPsi_Im_full,npadding,dt,alpha)
  return (f, fft_Psi)

def get_Psi_averaged_mean(data,alpha,npadding,M,sim_name):
  """ here using <\tilde{Psi}> = 1/4\pi \int \vert \tilde{\Psi} \vert \dd \Omega } """
  """ i.e. the mean abs """
  Nphi = 64
  Ntheta = 64
  out_filename = sim_name + "_Psi_fft_averaged_mean_Nphi{:d}_Ntheta{:d}".format(Nphi,Ntheta)
  try:
    #if (1==1):
    out_data = np.load(out_filename + ".npy")
    f = out_data[:,0]
    fft_Psi_out = out_data[:,1]
  except:
    r_ext = data[0,-4]
    r_ext_star = r_ext + 2*M*np.log((r_ext/(2.0*M)) - 1.0)
    data = data[data[:,0]>=r_ext_star,:]
    #
    phi_arr = np.linspace(0.0,2*np.pi,Nphi)
    cos_theta_arr = np.linspace(-1.0,1.0,Ntheta)
    theta_arr = np.arccos(cos_theta_arr)
    dphi = phi_arr[1] - phi_arr[0]
    dcos_theta = cos_theta_arr[1] - cos_theta_arr[0]
    f, fft_Psi = get_fft_Psi_full(data,0.0,0.0,alpha,npadding)
    fft_Psi_out = np.zeros(len(fft_Psi))
    ncount = 0
    for iphi in range(0,len(phi_arr)):
      for jtheta in range(0,len(theta_arr)):
        f, fft_Psi = get_fft_Psi_full(data,theta_arr[jtheta],phi_arr[iphi],alpha,npadding)
        fft_Psi_out += dphi*dcos_theta*np.abs(fft_Psi)
        ncount += 1
        print("\r>> integrated {:.3f}%".format(100.0*ncount/(Nphi*Ntheta)),end="",flush=True)
    fft_Psi_out = fft_Psi_out/(4*np.pi)
    out_data = np.column_stack((f, fft_Psi_out))
    np.save(out_filename,out_data)
  return (f[3:], fft_Psi_out[3:])

def get_Psi_averaged_rms(data,npadding,dt,alpha,M):
  """ here using <\tilde{Psi}> = sqrt{ 1/4\pi \int \vert \tilde{\Psi} \vert^2 \dd \Omega } """
  """ i.e. the rms """
  r_ext = data[0,-4]
  r_ext_star = r_ext + 2*M*np.log((r_ext/(2.0*M)) - 1.0)
  data = data[data[:,0]>=r_ext_star,:]
  Nt = len(data[:,0])+2*npadding
  N_max = int(Nt/2.0)
  rPsi_sum_of_modes = np.zeros(N_max-2)
  r_ext = data[0,-4]
  n_mode = 0
  for ll in [2,3,4,5,6,7,8]:
    for im in range(0,2*ll+1):
      mm = ll - im
      rPsi_Re = data[:,1+2*n_mode]
      rPsi_Im = data[:,2+2*n_mode]
      f, fft_Psi = get_fft(rPsi_Re,rPsi_Im,npadding,dt,alpha)
      rPsi_sum_of_modes += np.abs(fft_Psi)**2/(4*np.pi)
      n_mode += 1
  return (f[3:],np.sqrt(rPsi_sum_of_modes[3:])*r_ext)

def get_hphc_averaged_full(data,npadding,dt,alpha,M):
  r_ext = data[0,-4]
  r_ext_star = r_ext + 2*M*np.log((r_ext/(2*M)) - 1.0)
  data = data[data[:,0]>=r_ext_star,:]
  Nt = len(data[:,0])+2*npadding
  N_max = int(Nt/2.0)
  rhphc_sum_of_modes = np.zeros(N_max-2)
  n_mode = 0
  for ll in [2,3,4,5,6,7,8]:
    for im in range(0,2*ll+1):
      mm = ll - im
      rh_plus = data[:,1+2*n_mode]
      rh_cross = data[:,2+2*n_mode]
      f, fft_hphc = get_fft(rh_plus,rh_cross,npadding,dt,alpha)
      rhphc_sum_of_modes += np.abs(fft_hphc)**2/(4*np.pi)
      n_mode += 1
  return (f,np.sqrt(rhphc_sum_of_modes))

def get_hc_curve(sim_name,r_index,m_BH_in_MSun,Mpc_dist):

  dis = Mpc_dist * mpc_in_MSun
  z = get_z_from_luminosity_d(Mpc_dist)

  #print("len(Psi_data = ", len(Psi_data[:,0]))
  m_BH = 0.5
  f_max = 1.0/(4*np.pi*m_BH) 
  #(np.sqrt(27)*2*m_BH*2*np.pi) # using 2 pi QNM_freq ~ 1/(sqrt(27)M)
  if use_mean_or_rms:
    f_Psi, fft_Psi = get_Psi_averaged_mean(Psi_data,alpha,npadding,M,sim_name)
  else:
    f_Psi, fft_Psi = get_Psi_averaged_rms(Psi_data,npadding,dt,alpha,M)

  #
  fft_Psi = fft_Psi[f_Psi<f_max]

  #fft_hphc = fft_hphc[f_hphc<f_max]
  #f_hphc = f_hphc[f_hphc<f_max]
  #
  #print("f_Psi[0] = ", f_Psi[0])
  #print("f_hphc[0] = ", f_hphc[0])
  omega_Psi = 2*np.pi*f_Psi
  fft_Psi /= omega_Psi**2
  #
  code_units_to_m0 = 1/m_BH
  code_units_to_MSun = code_units_to_m0 * m_BH_in_MSun

  hc = hc_scaleless * m_BH_in_MSun * (1.0+z) / dis
  print("fM at max h_c = ", M*f_Psi[np.argmax(hc)])
  #
  f_in_Hz_scaleless = f_Psi/(MSun_in_s*(1.0+z))
  f_in_Hz = f_in_Hz_scaleless/code_units_to_MSun
  return (f_in_Hz,hc,ASD_Psi)

  f_detector, ASD_detector = detector_dict[detector_name]
  y_detector = np.log10(ASD_detector)
  f, hc, ASD = get_hc_curve(sim_name,4,m_BH_in_MSun,Mpc_dist)
  f_min = np.max((np.min(f),np.min(f_detector)))
  f_max = np.min((np.max(f),np.max(f_detector)))
  ASD = ASD[(f>=f_min)*(f<=f_max)]
  f = f[(f>=f_min)*(f<=f_max)]
  y_data = np.log10(ASD)
  y_detector_interp = np.interp(f,f_detector,y_detector)
  integrand = 10**(2*(y_data - y_detector_interp))
  log_f = np.log(f)
  integral = integrate.simps(integrand,log_f)
  SNR = np.sqrt(integral)
  return SNR
